```python
# scripts/blender/bake_w1_textures.py
# Run: blender --background --python scripts/blender/bake_w1_textures.py
# hand-painted stylized 1024 maps. No photo sources. Deterministic (seed 7).
import bpy, os
import numpy as np
PROJ = r"C:\anki\スケジュール管理アプリ"
OUT = os.path.join(PROJ, "artifacts", "3d", "textures", "w1")
os.makedirs(OUT, exist_ok=True)
S, SEED = 1024, 7
rng = np.random.default_rng(SEED)

def fbm(shape, octs=4):
    h, w = shape
    acc = np.zeros(shape, np.float64); amp, f = 1.0, 4
    for _ in range(octs):
        gh, gw = max(1, h // (S // f)), max(1, w // (S // f))
        g = rng.random((gh + 1, gw + 1))
        yy = np.linspace(0, gh, h)[:, None]; xx = np.linspace(0, gw, w)[None, :]
        y0, x0 = yy.astype(int).clip(0, gh - 1), xx.astype(int).clip(0, gw - 1)
        y1, x1 = (y0 + 1).clip(0, gh), (x0 + 1).clip(0, gw)
        fy, fx = (yy - y0), (xx - x0)
        sy, sx = fy * fy * (3 - 2 * fy), fx * fx * (3 - 2 * fx)
        acc += amp * (g[y0, x0] * (1 - sx) * (1 - sy) + g[y0, x1] * sx * (1 - sy)
                      + g[y1, x0] * (1 - sx) * sy + g[y1, x1] * sx * sy)
        amp *= 0.5; f *= 2
    return (acc - acc.min()) / (acc.max() - acc.min() + 1e-9)

def height_to_normal(h, strength=2.0):
    dy, dx = np.gradient(h)
    n = np.stack([-dx * strength, -dy * strength, np.ones_like(h)], -1)
    return (n / np.linalg.norm(n, axis=-1, keepdims=True) * 0.5 + 0.5)

def save(name, arr, srgb):
    arr = (np.clip(arr, 0, 1) * 255).astype(np.uint8)
    if arr.ndim == 2:
        arr = np.stack([arr] * 3 + [np.full_like(arr, 255)], -1)
    elif arr.shape[2] == 3:
        arr = np.concatenate([arr, np.full(arr.shape[:2] + (1,), 255, np.uint8)], -1)
    img = bpy.data.images.new(name, S, S, alpha=True)
    img.pixels = (arr.astype(np.float32) / 255.0).ravel().tolist()
    img.file_format = 'PNG'
    img.filepath_raw = os.path.join(OUT, name + ".png")
    img.save()
    bpy.data.images.remove(img)

# --- rock: horizontal strata + cracks + bumps (reddish sandstone) ---
strata = 0.5 + 0.5 * np.sin(np.linspace(0, 1, S)[:, None] * 28 + fbm((S, S)) * 6)
cracks = (fbm((S, S), 5) > 0.72).astype(float) * 0.8
bump = fbm((S, S), 4)
rh = np.clip(0.55 * strata + 0.45 * bump - cracks * 0.5, 0, 1)
base = np.stack([
    np.clip(0.62 + strata * 0.16 - cracks * 0.25 + (bump - 0.5) * 0.12, 0, 1),
    np.clip(0.38 + strata * 0.10 - cracks * 0.18 + (bump - 0.5) * 0.10, 0, 1),
    np.clip(0.27 + strata * 0.07 - cracks * 0.12 + (bump - 0.5) * 0.08, 0, 1)], -1)
save("rock_base", base, True)
save("rock_normal", height_to_normal(rh, 2.0), False)
save("rock_rough", np.clip(0.75 + (bump - 0.5) * 0.3 + cracks * 0.15, 0, 1), False)
save("rock_ao", np.clip(0.65 + rh * 0.35, 0, 1), False)

# --- grass: tone variation + moss + blade pattern ---
tone = fbm((S, S), 4)
moss = (fbm((S, S), 3) > 0.60).astype(float)
blades = 0.5 + 0.5 * np.sin(np.linspace(0, 1, S)[None, :] * 220 + tone * 9)
gh = np.clip(tone * 0.6 + blades * 0.25 + moss * 0.15, 0, 1)
gbase = np.stack([
    np.clip(0.30 + tone * 0.22 - moss * 0.10 + (blades - 0.5) * 0.10, 0, 1),
    np.clip(0.55 + tone * 0.20 + moss * 0.05 + (blades - 0.5) * 0.08, 0, 1),
    np.clip(0.22 + tone * 0.12 - moss * 0.02 + (blades - 0.5) * 0.06, 0, 1)], -1)
save("grass_base", gbase, True)
save("grass_normal", height_to_normal(gh, 1.2), False)
save("grass_rough", np.clip(0.85 + (tone - 0.5) * 0.2 - moss * 0.1, 0, 1), False)
save("grass_ao", np.clip(0.70 + gh * 0.30, 0, 1), False)
print("BAKE_W1_TEXTURES_OK")
```

`diorama_kit.py` に以下を作成する（bevelは `bmesh.ops.bevel` 幅0.08・2セグメント、頂点色は `mesh.color_attributes` の `Color` に岩系3色のばらつき）：

```python
# scripts/blender/diorama_kit.py (Task 2 分)
import bpy, bmesh, math
from mathutils import Vector

def mat_textured(name, base_img, normal_img, rough_img, ao_img, uv_scale):
    m = bpy.data.materials.get(name) or bpy.data.materials.new(name=name)
    m.use_nodes = True
    bsdf = m.node_tree.nodes.get("Principled BSDF")
    bsdf.inputs["Base Color"].default_value = (1, 1, 1, 1)
    bsdf.inputs["Roughness"].default_value = 1.0
    tc = m.node_tree.nodes.new('ShaderNodeTexCoord')
    mp = m.node_tree.nodes.new('ShaderNodeMapping')
    mp.inputs['Scale'].default_value = (uv_scale, uv_scale, 1.0)
    def tex(img, space):
        t = m.node_tree.nodes.new('ShaderNodeTexImage')
        t.image = bpy.data.images.load(img)
        t.image.colorspace_settings.name = space
        try: t.image.pack()
        except RuntimeError: pass
        m.node_tree.links.new(tc.outputs['UV'], mp.inputs['Vector'])
        return t
    cb, nb, rb, ab = tex(base_img, 'sRGB'), tex(normal_img, 'Non-Color'), tex(rough_img, 'Non-Color'), tex(ao_img, 'Non-Color')
    m.node_tree.links.new(mp.outputs['Vector'], cb.inputs['Vector'])
    m.node_tree.links.new(mp.outputs['Vector'], nb.inputs['Vector'])
    m.node_tree.links.new(mp.outputs['Vector'], rb.inputs['Vector'])
    m.node_tree.links.new(mp.outputs['Vector'], ab.inputs['Vector'])
    nm = m.node_tree.nodes.new('ShaderNodeNormalMap')
    nm.inputs['Strength'].default_value = 0.8
    m.node_tree.links.new(nb.outputs['Color'], nm.inputs['Color'])
    m.node_tree.links.new(nm.outputs['Normal'], bsdf.inputs['Normal'])
    m.node_tree.links.new(cb.outputs['Color'], bsdf.inputs['Base Color'])
    m.node_tree.links.new(rb.outputs['Color'], bsdf.inputs['Roughness'])
    ao = m.node_tree.nodes.new('ShaderNodeMix')
    ao.data_type = 'RGBA'; ao.blend_type = 'MULTIPLY'
    ao.inputs[0].default_value = 1.0
    ao.inputs[7].default_value = (1, 1, 1, 1)
    m.node_tree.links.new(cb.outputs['Color'], ao.inputs[6])
    m.node_tree.links.new(ab.outputs['Color'], ao.inputs[7])
    m.node_tree.links.new(ao.outputs[2], bsdf.inputs['Base Color'])
    return m
```
